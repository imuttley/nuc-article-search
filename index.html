Document Search WebComponent

Integrable on any CMS via <script> tag. No React required by the host page.

Integration Examples 
<!-- 1. Add script tag (in <head> or before </body>) -->
<script type="module" src="document-search.js"></script>

<!-- 2. Place the element anywhere in your HTML -->
<document-search
  api-url="https://your-api.com/search"
  placeholder="Cerca..."
  page-size="10"
  debounce-ms="300"
></document-search>

<!-- 3. Programmatic config (optional) -->
<script>
  const el = document.querySelector('document-search');
  el.searchHeaders = { Authorization: 'Bearer YOUR_TOKEN' };
  el.addEventListener('results', e => console.log('Results:', e.detail));
  el.addEventListener('error', e => console.error('Error:', e.detail));
</script>
